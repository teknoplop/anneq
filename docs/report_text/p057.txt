                fprintf(output_file,"%+2.15lf\n", errors[i]/runs);
                }

        fclose(output_file);

        if ((output_file = fopen("e:\\temp\\ber.txt", "wt")) == NULL)
                {
                printf("File ber.txt cannot be created");
                exit(1);
                }


        fprintf(output_file,"%+2.15lf\n", ber/runs);
